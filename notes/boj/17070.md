못품.
중복으로 dfs 랑 dp 를 사용했음.
입력을 이상하게 했음.
어차피 y가 증가, 그리고 그안에 x 증가로
반복문으로 풀면 쉽다.
지금 가장 중요한 문제는 추후에 dfs 가 늦게 도착할 경우
'또다시' dfs 를 실행한다는 것이다.


아래는 못푼코드 전문

```python
import sys
# sys.stdin = open("input.txt", "r")
input = sys.stdin.readline
N = int(input())
board = [list(map(int, input().split())) for _ in range(N)]

# 밀기, 회전은 45 도만 가능.

# dp  경우의 숫자 합치기로.
# 회전하면 상태 변경
# 상태 가로 는 0 대각선은 1 세로는 2

dp = [list([0] * 3 for col in range(N)) for row in range(N)]


dp[0][1][0] = 1
def check(y, x):
    return (0 <= y < N and 0 <= x < N)

def check_move(ny, nx, cur_state, new_state):
    if cur_state == 0:
        if new_state == 0:
            if board[ny][nx]:
                return False
        elif new_state == 1:
            if board[ny][nx] or board[ny - 1][nx] or board[ny][nx - 1]:
                return False
    elif cur_state == 1:
        if new_state == 0:
            if board[ny][nx]:
                return False
        elif new_state == 1:
            if board[ny][nx] or board[ny - 1][nx] or board[ny][nx - 1]:
                return False
        elif new_state == 2:
            if board[ny][nx]:
                return False
    elif cur_state == 2:
        if new_state == 2:
            if board[ny][nx]:
                return False
        elif new_state == 1:
            if board[ny][nx] or board[ny - 1][nx] or board[ny][nx - 1]:
                return False
    return True


def dfs(y, x, cur_state):
    cy = y
    cx = x
    if cur_state == 0:
        # 그냥 오른쪽
        new_state = 0
        ny = cy
        nx = cx + 1
        if check(ny, nx) and check_move(ny, nx, cur_state, new_state):
            dp[ny][nx][new_state] += dp[y][x][cur_state]
            dfs(ny, nx, new_state)
        # 오른쪽 이후 회전
        new_state = 1
        ny = cy + 1
        nx = cx + 1
        if check(ny, nx) and check_move(ny, nx, cur_state, new_state):
            dp[ny][nx][new_state] += dp[y][x][cur_state]
            dfs(ny, nx, new_state)
    elif cur_state == 1:
        # 대각 아래
        new_state = 1
        ny = cy + 1
        nx = cx + 1
        if check(ny, nx) and check_move(ny, nx, cur_state, new_state):
            dp[ny][nx][new_state] += dp[y][x][cur_state]
            dfs(ny, nx, new_state)
        # 위쪽
        new_state = 0
        ny = cy
        nx = cx + 1
        if check(ny, nx) and check_move(ny, nx, cur_state, new_state):
            dp[ny][nx][new_state] += dp[y][x][cur_state]
            dfs(ny, nx, new_state)
        # 아래쪽
        new_state = 2
        ny = cy + 1
        nx = cx
        if check(ny, nx) and check_move(ny, nx, cur_state, new_state):
            dp[ny][nx][new_state] += dp[y][x][cur_state]
            dfs(ny, nx, new_state)

    elif cur_state == 2:
        # 아래
        new_state = 2
        ny = cy + 1
        nx = cx
        if check(ny, nx) and check_move(ny, nx, cur_state, new_state):
            dp[ny][nx][new_state] += dp[y][x][cur_state]
            dfs(ny, nx, new_state)
        # 아래 이후 대각선
        new_state = 1
        ny = cy + 1
        nx = cx + 1
        if check(ny, nx) and check_move(ny, nx, cur_state, new_state):
            dp[ny][nx][new_state] += dp[y][x][cur_state]
            dfs(ny, nx, new_state)
dfs(0, 1, 0)
ret = 0
for i in range(3):
    ret += dp[N - 1][N - 1][i]

for i in dp:
    print(i)
print(ret)
```
