2. 시간복잡도를 제대로 계산해내지 못하고 bfs 말고 인접한 배열을 체크하는 로직을 떠올리지 못함.

# 0번 - 1번 구조물
if not check_side(
    [h0, h0 + l0 - 1],
    [h1, h1 + l1 - 1]

    continue

# 1번 - 2번 구조물

    [h1, h1 + l1 - 1],
