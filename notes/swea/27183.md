```python
# 최대 값으로 노드 초기화
def build(s, e, node):
    if s == e:
        tree[node] = arr[s]
        return tree[node]

    mid = (s + e) // 2

    tree[node] = max(
        build(s, mid, node * 2),
        build(mid + 1, e, node * 2 + 1)
    )

    return tree[node]

# 노드 특정값 변환
def set_node(node, value):
    tree[node] = value
    lazy_set[node] = value
    lazy_add[node] = 0

# 노드에 특정값 더해주기
def add_node(node, value):
    tree[node] += value
	# lazy set 이 none 이 아니면 lazy set 에 더해줘야함
    if lazy_set[node] is not None:
        lazy_set[node] += value
    # none 이면 add 에 갱신
    else:
        lazy_add[node] += value

# 
def push_down(s, e, node):
    if s == e:
        return

    lc = node * 2
    rc = node * 2 + 1

    if lazy_set[node] is not None:
        set_node(lc, lazy_set[node])
        set_node(rc, lazy_set[node])
        lazy_set[node] = None

    if lazy_add[node] != 0:
        add_node(lc, lazy_add[node])
        add_node(rc, lazy_add[node])
        lazy_add[node] = 0


def update_set(s, e, node, l, r, value):
    if r < s or e < l:
        return

    if l <= s and e <= r:
        set_node(node, value)
        return

    push_down(s, e, node)

    mid = (s + e) // 2

    update_set(s, mid, node * 2, l, r, value)
    update_set(mid + 1, e, node * 2 + 1, l, r, value)

    tree[node] = max(
        tree[node * 2],
        tree[node * 2 + 1]
    )


def update_add(s, e, node, l, r, value):
    if r < s or e < l:
        return

    if l <= s and e <= r:
        add_node(node, value)
        return

    push_down(s, e, node)

    mid = (s + e) // 2

    update_add(s, mid, node * 2, l, r, value)
    update_add(mid + 1, e, node * 2 + 1, l, r, value)

    tree[node] = max(
        tree[node * 2],
        tree[node * 2 + 1]
    )


def query_max(s, e, node, l, r):
    if r < s or e < l:
        return float('-inf')

    if l <= s and e <= r:
        return tree[node]

    push_down(s, e, node)

    mid = (s + e) // 2

    left_val = query_max(
        s, mid, node * 2,
        l, r
    )

    right_val = query_max(
        mid + 1, e, node * 2 + 1,
        l, r
    )

    return max(left_val, right_val)


def query_first(s, e, node, l, r, x):
    if r < s or e < l:
        return -1

    if tree[node] < x:
        return -1

    if s == e:
        return s

    push_down(s, e, node)

    mid = (s + e) // 2

    left_result = query_first(
        s, mid, node * 2,
        l, r, x
    )

    if left_result != -1:
        return left_result

    right_result = query_first(
        mid + 1, e, node * 2 + 1,
        l, r, x
    )

    return right_result


def init(N, fatigue):
    global g_N, arr, tree, lazy_add, lazy_set

    g_N = N
    arr = fatigue

    tree = [0] * (4 * N)
    lazy_add = [0] * (4 * N)
    lazy_set = [None] * (4 * N)

    build(0, g_N - 1, 1)


def add_stress(l, r, w):
    update_add(0, g_N - 1, 1, l, r, w)


def repair(l, r, v):
    update_set(0, g_N - 1, 1, l, r, v)


def get_peak(l, r):
    return query_max(0, g_N - 1, 1, l, r)


def find_risk(l, r, x):
    return query_first(0, g_N - 1, 1, l, r, x)


def input_tokens():
    while True:
        try:
            line = input()
        except EOFError:
            return

        for token in line.split():
            yield token


def main():
    it = input_tokens()

    T = int(next(it))

    for tc in range(1, T + 1):
        N = int(next(it))
        M = int(next(it))

        fatigue = [int(next(it)) for _ in range(N)]

        init(N, fatigue)

        result = [f'#{tc}']

        for _ in range(M):
            op = int(next(it))

            if op == 1:
                l = int(next(it))
                r = int(next(it))
                w = int(next(it))
                add_stress(l, r, w)

            elif op == 2:
                l = int(next(it))
                r = int(next(it))
                v = int(next(it))
                repair(l, r, v)

            elif op == 3:
                l = int(next(it))
                r = int(next(it))
                result.append(str(get_peak(l, r)))

            else:
                l = int(next(it))
                r = int(next(it))
                x = int(next(it))
                result.append(str(find_risk(l, r, x)))

        print('\n'.join(result))


main()
```
