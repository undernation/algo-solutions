from typing import List
from collections import deque
import heapq
class RESULT:
    def __init__(self):
        self.cnt = 0
        self.IDs = [0, 0, 0, 0, 0]

class Worm:
    def __init__(self):
        self.worm_id = -1
        self.is_alive = True
        self.head_pos = [-1, -1]
        self.tails = deque()
        self.direction = 3
        self.version = 0
        self.length = -1
        self.poten = 0

def is_straight(worm):
    hy, hx = worm.head_pos
    ty, tx = worm.tails[-1]

    return hy == ty or hx == tx

def move(worm_id):
    cur_worm = worms[worm_id]

    # 일직선이면 시계 방향 회전
    if is_straight(cur_worm):
        cur_worm.direction = (cur_worm.direction + 1) % 4

    dy, dx = dydx[cur_worm.direction]

    cy, cx = cur_worm.head_pos

    # 기존 머리는 이제 몸통
    cur_worm.tails.appendleft((cy, cx))

    # 새 머리 위치
    ny = cy + dy
    nx = cx + dx

    cur_worm.head_pos[0] = ny
    cur_worm.head_pos[1] = nx

def remove_worm(worm_id):
    cur_worm = worms[worm_id]

    cur_worm.is_alive = False
    alive_ids.remove(worm_id)

    # 현재 head는 아직 board에 등록되지 않았음
    # tails만 board에 존재
    for pos in cur_worm.tails:
        del board[pos]

    cur_worm.tails.clear()
def is_head(worm_id, cy, cx):
    hy, hx = worms[worm_id].head_pos

    if hy == cy and hx == cx:
        return True

    return False

def proceed(time):
    global g_time

    while g_time < time:

        dead = set()
        collisions = []

        # 이번 초 새 머리 위치
        head_owner = {}

        # =====================================
        # 1. 모든 지렁이 이동
        # =====================================
        for worm_id in alive_ids:
            move(worm_id)

            worm = worms[worm_id]
            hy, hx = worm.head_pos

            # 격자 밖
            if not (0 <= hy < g_N and 0 <= hx < g_N):
                dead.add(worm_id)
                continue

            pos = (hy, hx)

            # ---------------------------------
            # 머리 머리 충돌
            # ---------------------------------
            other = head_owner.get(pos)

            if other is None:
                head_owner[pos] = worm_id

            else:
                dead.add(worm_id)
                dead.add(other)

        # =====================================
        # 2. 머리 몸통 충돌
        # =====================================
        for worm_id in alive_ids:

            worm = worms[worm_id]
            hy, hx = worm.head_pos

            if not (0 <= hy < g_N and 0 <= hx < g_N):
                continue

            pos = (hy, hx)

            # board에는 이동이 끝난 "몸통"만 존재
            target = board.get(pos)

            # 자기 몸은 제외
            if target is not None and target != worm_id:
                dead.add(worm_id)
                collisions.append((worm_id, target))

        # =====================================
        # 3. 성장 잠재력 지급
        # =====================================
        for attacker, target in collisions:

            # target도 이번에 죽으면 potential 못 받음
            if target in dead:
                continue

            worms[target].poten += worms[attacker].length

        # =====================================
        # 4. 죽은 지렁이 제거
        # =====================================
        for worm_id in dead:
            remove_worm(worm_id)

        # =====================================
        # 5. 살아남은 새 머리를 board에 등록
        # =====================================
        for worm_id in alive_ids:
            worm = worms[worm_id]

            hy, hx = worm.head_pos

            board[(hy, hx)] = worm_id

        g_time += 1

def top5(mTime: int) -> RESULT:
    proceed(mTime)

    ret = RESULT()

    cnt = 0
    removed = []

    while hq:
        cur_length, neg_id, cur_version = heapq.heappop(hq)
        cur_id = -neg_id

        cur_worm = worms[cur_id]

        # stale
        if cur_version != cur_worm.version:
            continue

        # 죽은 지렁이
        if not cur_worm.is_alive:
            continue

        ret.IDs[cnt] = cur_id
        cnt += 1

        # tuple
        removed.append((cur_length, neg_id, cur_version))

        if cnt == 5:
            break

    ret.cnt = cnt

    # tuple 그대로
    for item in removed:
        heapq.heappush(hq, item)

    return ret
