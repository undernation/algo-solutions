```python
from typing import List
from collections import deque
import heapq


class RESULT:
    def __init__(self, mHeights: List[List[int]]):
        self.heights = mHeights


DIR = ((-1, 0), (0, 1), (1, 0), (0, -1))

g_N = -1
board = []
direction_board = []


def bfs(sy, sx, visited):
    visited[sy][sx] = True
    q = deque([(sy, sx)])
    cells = []

    while q:
        y, x = q.popleft()
        cells.append((y, x))

        for dy, dx in DIR:
            ny, nx = (y + dy) % g_N, (x + dx) % g_N

            if board[ny][nx] > 0 and not visited[ny][nx]:
                visited[ny][nx] = True
                q.append((ny, nx))

    return cells


def melt():
    targets = []

    for y in range(g_N):
        for x in range(g_N):
            if board[y][x] == 0:
                continue

            for dy, dx in DIR:
                ny, nx = (y + dy) % g_N, (x + dx) % g_N

                if board[ny][nx] == 0:
                    targets.append((y, x))
                    break
    for y, x in targets:
        board[y][x] -= 1


def move():
    global board, direction_board

    new_board = [[0] * g_N for _ in range(g_N)]
    best = [[None] * g_N for _ in range(g_N)]
    visited = [[False] * g_N for _ in range(g_N)]

    for y in range(g_N):
        for x in range(g_N):
            if board[y][x] == 0 or visited[y][x]:
                continue

            cells = bfs(y, x, visited)
            volume = sum(board[cy][cx] for cy, cx in cells)
            d = direction_board[y][x]

            priority = (-volume, len(cells), y, x, d)
            dy, dx = DIR[d]

            for cy, cx in cells:
                ny, nx = (cy + dy) % g_N, (cx + dx) % g_N

                new_board[ny][nx] = max(
                    new_board[ny][nx], board[cy][cx]
                )
                if best[ny][nx] is None or priority < best[ny][nx]:
                    best[ny][nx] = priority

    board = new_board
    direction_board = [[-1] * g_N for _ in range(g_N)]
    visited = [[False] * g_N for _ in range(g_N)]

    for y in range(g_N):
        for x in range(g_N):
            if board[y][x] == 0 or visited[y][x]:
                continue

            cells = bfs(y, x, visited)
            winner = min(best[cy][cx] for cy, cx in cells)
            d = winner[-1]

            for cy, cx in cells:
                direction_board[cy][cx] = d


def init(N: int, M: int, mIceBlock: List[List[int]], mIceGroup: List[List[int]]) -> None:
    global g_N, board, direction_board
    g_N = N
    board = [[0] * g_N for _ in range(g_N)]
    direction_board = [[-1] * g_N for _ in range(g_N)]
    visited = [[False] * N for _ in range(N)]
    board = mIceBlock

    for i in range(M):
        x, y, d = mIceGroup[i]

        for cy, cx in bfs(y, x, visited):
            direction_board[cy][cx] = d


def oneYearLater() -> RESULT:
    melt()
    move()
    res = RESULT([[0 for _ in range(100)] for _ in range(100)])
    for i in range(g_N):
        for j in range(g_N):
            res.heights[i][j] = board[i][j]
    return res
```
